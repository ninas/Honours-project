% ---------------------------------------  Experiment ---------------------------------------------

\newglossaryentry{c5:ecolo}{
name={Ecological validity},
description={Refers to how closely the experimental conditions approximate real-life situations and thus the degree to which the findings can be generalised outside of an experimental setting}}

\newglossaryentry{c5:between}{
name={Between-subjects design},
description={The use of distinct and mutually exclusive groups where each group either possesses or is exposed to only one experimental condition}}

\newglossaryentry{c5:within}{
name={Within-subjects design},
description={See Repeated measures}}

\newglossaryentry{c5:repeated}{
name={Repeated measures design},
description={Also known as a Within-subjects design. The use of a single group exposed to multiple experimental conditions in order to assess their effect on the dependent variable(s)}}

\newglossaryentry{c5:likert}{
name={Likert scale},
description={A rating scale where individuals rate their level of agreement with a statement}}

\newglossaryentry{c5:quota}{
name={Quota sampling},
description={The non-probabilistic selection of participants based on specific subject characteristics deemed important to the topic under investigation}}

\newglossaryentry{c5:order}{
name={Order effect},
description={Caveat associated with the use of a repeated measures design. The dependent variable is affected by the order in which conditions are presented}}

\newglossaryentry{c5:fatig}{
name={Fatigue effect},
description={Caveat associated with the use of a repeated measures design. This describes a deterioration in performance as the research participant becomes tired, bored or distracted}}

\newglossaryentry{c5:contrE}{
name={Contrast effect},
description={Caveat associated with the use of a repeated measures design. This describes a condition where the response to the second or third condition in an experiment is altered because the two conditions are contrasted with each other}}

\newglossaryentry{c5:practice}{
name={Practice effect},
description={Caveat associated with the use of a repeated measures design. This describes a condition where performance on a task improves as a result of repeated practice with the task}}

\newglossaryentry{c5:counterB}{
name={Counterbalancing},
description={A method of controlling for order effects in a repeated measures design by either testing all possible combinations or randomising the order in which experimental conditions are completed}}

\newglossaryentry{c5:exprB}{
name={Experimenter bias},
description={Also termed experimenter expectancy. Any intentional or unintentional influence the experimenter exerts on participants in order to confirm the hypothesis under investigation}}

\newglossaryentry{c5:singB}{
name={Single-blind study},
description={Experimental design where participants are unaware of which experimental condition they have been assigned to. However, the experimenter has full knowledge of this information}}

\newglossaryentry{c5:doublB}{
name={Double-blind study},
description={Experimental design where neither the experimenter nor participants are aware of which experimental conditions the participants have been assigned to}}

\newglossaryentry{c5:demandC}{
name={Demand characteristics},
description={Any feature of the experiment that may inform participants as to the purpose of the study}}

\newglossaryentry{c5:MANOVA}{
name={MANOVA},
description={Multivariate Analysis of Variance. Generalisation of the univariate ANOVA. Used to analyse effects in studies containing more than one dependent variable. See also: ANOVA}}

\newglossaryentry{c5:Mauch}{
name={Mauchly's sphericity test},
description={Used as a test of the sphericity of data in repeated measures designs. Sphericity refers to the equality of the variances between different levels of the repeated measures independent variable. Sphericity is an assumption of a repeated measures ANOVA and thus deviations may cause invalid results. See also: ANOVA, MANOVA, Repeated measures}}

\newglossaryentry{c5:fisher}{
name={Fisher LSD test},
description={Fisher Least Significant Difference test. Provides post-hoc testing on the results of an ANOVA by performing pairwise comparisons of the means}}

\newglossaryentry{c5:green}{
name={Greenhouse-Geisser correction},
description={Correction used when assumptions of sphericity are violated for repeated measures ANOVAs.  In order to produce more accurate significance values, the degrees of freedom of the ANOVA are adjusted. See also: Mauchly's sphericity test, ANOVA, MANOVA, Repeated measures}}

\newglossaryentry{c5:wilks}{
name={Wilks' lambda distribution},
description={Multivariate generalisation of the \emph{F}-distribution. Used by the MANOVA procedure in order to determine significance values. See also: MANOVA}}

\newglossaryentry{c5:ordinal}{
name={Ordinal interaction},
description={This describes an interaction that has simple effects that lie in the same direction. See also: Interaction (statistics)}}

\newglossaryentry{c5:disordinal}{
name={Disordinal interaction},
description={This describes an interaction that has simple effects that lie in opposite directions. See also: Interaction (statistics)}}

\newglossaryentry{c5:multiCon}{
name={Multivariate contrast analysis},
description={This is a procedure that provides a linear combination of two or more factor level means, that is, it allows one to collapse these means into one dimension, in order to compare specific elements of an experimental design}}
